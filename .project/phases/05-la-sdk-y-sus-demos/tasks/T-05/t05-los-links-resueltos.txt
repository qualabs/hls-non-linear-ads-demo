# Los links de los dos README, resueltos contra el disco: primero los links
# markdown, y despues cada ruta escrita entre backticks o dentro de un bloque de
# codigo, que es la forma en que una ruta se quedo vieja dos veces en esta fase.
import re, sys
from pathlib import Path

ROOT = Path('.').resolve()
FILES = ['README.md', 'demo/compatibility-pair/README.md']
LINK = re.compile(r'\[[^\]]*\]\(([^)]+)\)')
SPAN = re.compile(r'`([^`\n]+)`')
TOKEN = re.compile(r'[A-Za-z0-9_./<>-]+')
EXTS = ('.js', '.mjs', '.sh', '.json', '.md', '.html', '.css', '.svg')
missing = 0

def looks_like_path(tok):
    if ' ' in tok or tok.startswith('/') or '://' in tok or '<' in tok:
        return False
    return tok.endswith('/') or tok.endswith(EXTS)

print('== links markdown ==')
for f in FILES:
    p = ROOT / f
    targets = LINK.findall(p.read_text())
    print(f'--- {f}: {len(targets)} links')
    for t in targets:
        if t.startswith(('http://', 'https://', 'mailto:', '#')):
            print(f'    absoluto  {t}')
            continue
        resolved = (p.parent / t.split('#')[0]).resolve()
        ok = resolved.exists()
        missing += 0 if ok else 1
        print(f'    {"OK      " if ok else "FALTA   "} {t}  ->  {resolved.relative_to(ROOT)}')

print()
print('== rutas escritas en prosa y en los bloques de codigo ==')
for f in FILES:
    p = ROOT / f
    prose, fences, in_fence = [], [], False
    for line in p.read_text().split('\n'):
        if line.startswith('```'):
            in_fence = not in_fence
            continue
        (fences if in_fence else prose).append(line)
    cands = set(SPAN.findall('\n'.join(prose))) | set(TOKEN.findall('\n'.join(fences)))
    print(f'--- {f}')
    for tok in sorted(c for c in cands if looks_like_path(c)):
        cand = tok[2:] if tok.startswith('./') else tok
        base = ('relativa al propio archivo' if (p.parent / cand).exists()
                else ('relativa a la raiz del repo' if (ROOT / cand).exists() else None))
        missing += 0 if base else 1
        print(f'    {"OK    " if base else "FALTA "} {tok:56s} {base or "NO EXISTE en ninguna de las dos bases"}')

print()
print(f'{missing} ruta(s) sin resolver')
sys.exit(1 if missing else 0)
