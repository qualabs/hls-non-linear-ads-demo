# Las rutas escritas en TODO el arbol tracked (fuera de .project/), resueltas
# contra el disco: los links markdown, y ademas cada ruta escrita en prosa, en un
# bloque de codigo o en un comentario de codigo -- que es la forma en que una ruta
# se quedo vieja dos veces en esta fase. Extiende el chequeo de la T-05, que
# miraba solo los dos README.
import re, subprocess, sys
from pathlib import Path

ROOT = Path('.').resolve()
TEXT = ('.md', '.js', '.mjs', '.sh', '.html', '.css', '.json', '.svg')
tracked = subprocess.run(['git', 'ls-files'], capture_output=True, text=True).stdout.split()
FILES = [f for f in tracked if not f.startswith('.project/') and f.endswith(TEXT)]

LINK = re.compile(r'\[[^\]]*\]\(([^)]+)\)')
TOKEN = re.compile(r'[A-Za-z0-9_./<>-]+')
EXTS = ('.js', '.mjs', '.sh', '.json', '.md', '.html', '.css', '.svg')
missing = 0

def looks_like_path(tok):
    if ' ' in tok or tok.startswith('/') or '://' in tok or '<' in tok:
        return False
    return tok.endswith('/') or tok.endswith(EXTS)

def bases(p):
    # el propio directorio, la raiz del repo, y -- para lo que vive adentro de una
    # demo -- la raiz de esa demo, que es contra la que hacen `cd` sus scripts.
    out = [('relativa al propio archivo', p.parent), ('relativa a la raiz del repo', ROOT)]
    parts = p.relative_to(ROOT).parts
    if len(parts) > 2 and parts[0] == 'demo':
        out.insert(1, (f'relativa a demo/{parts[1]}/', ROOT / 'demo' / parts[1]))
    return out

print('== links markdown ==')
for f in FILES:
    if not f.endswith('.md'):
        continue
    p = ROOT / f
    targets = LINK.findall(p.read_text())
    if not targets:
        continue
    print(f'--- {f}: {len(targets)} links')
    for t in targets:
        if t.startswith(('http://', 'https://', 'mailto:', '#')):
            print(f'    absoluto  {t}')
            continue
        resolved = (p.parent / t.split('#')[0]).resolve()
        ok = resolved.exists()
        missing += 0 if ok else 1
        print(f'    {"OK      " if ok else "FALTA   "} {t}  ->  {resolved.relative_to(ROOT)}')

print()
print('== rutas escritas en prosa, en bloques de codigo y en comentarios ==')
for f in FILES:
    p = ROOT / f
    cands = {c for c in TOKEN.findall(p.read_text()) if looks_like_path(c)}
    if not cands:
        continue
    print(f'--- {f}')
    for tok in sorted(cands):
        cand = tok[2:] if tok.startswith('./') else tok
        base = next((name for name, b in bases(p) if (b / cand).exists()), None)
        missing += 0 if base else 1
        print(f'    {"OK    " if base else "FALTA "} {tok:56s} {base or "NO EXISTE en ninguna base"}')

print()
print(f'{missing} ruta(s) sin resolver')
sys.exit(1 if missing else 0)
